==================== CELL 3: auto-map class names ====================
import yaml, glob, collections

OVERRIDE = {}   # only if this cell says UNKNOWN, e.g. {"some name": "pole"} or {"some name": None}

def load_names(loc):
    n = yaml.safe_load(open(f"{loc}/data.yaml"))["names"]
    return [n[k] for k in sorted(n)] if isinstance(n, dict) else n

def auto_map(name):
    n = name.lower().strip()
    if n in OVERRIDE: return OVERRIDE[n]
    if any(k in n for k in ("telecom", "comm", "fiber", "fibre", "splice", "cable box", "terminal")): return "comm"
    if "transformer" in n or "xfmr" in n: return "transformer"
    if "cross" in n: return "crossarm"
    if "insulator" in n: return "insulator"
    if "pole" in n: return "pole"
    if any(k in n for k in ("wire", "line", "cable", "conductor")): return None
    return "UNKNOWN"

ORDER = ["pole", "transformer", "crossarm", "insulator", "comm"]
MAPS, USE, present = {}, [], set()
for i, loc in enumerate(dsets):
    names = load_names(loc)
    c = collections.Counter()
    for split in ("train", "valid", "test"):
        for f in glob.glob(f"{loc}/{split}/labels/*.txt"):
            for line in open(f):
                if line.strip():
                    c[names[int(line.split()[0])]] += 1
    MAPS[i] = [auto_map(nm) for nm in names]
    print(f"\nds{i} ({SOURCES[i][1]}):")
    for nm, mp in zip(names, MAPS[i]):
        tag = "DROPPED" if mp is None else ("UNKNOWN -> add to OVERRIDE" if mp == "UNKNOWN" else f"-> {mp}")
        print(f"   {nm:<25} {c[nm]:>6} boxes   {tag}")
    kept = {m for m in MAPS[i] if m not in (None, "UNKNOWN")}
    if kept:
        USE.append(i); present |= kept
        print(f"   USING this dataset (gives: {sorted(kept)})")
    else:
        print("   SKIPPING this dataset: it has none of the classes we need")

FINAL = [k for k in ORDER if k in present]
print("\nFINAL classes:", FINAL)
print("Datasets used:", [SOURCES[i][1] for i in USE])

==================== CELL 4: merge + make a real validation set ====================
import os, shutil, random
OUT = "/content/merged"
shutil.rmtree(OUT, ignore_errors=True)
counts = collections.Counter()

for i in USE:
    loc = dsets[i]
    for split in ("train", "valid", "test"):
        img_dir = f"{loc}/{split}/images"
        if not os.path.isdir(img_dir):
            continue
        dst = "valid" if split == "test" else split          # use test photos for grading too
        os.makedirs(f"{OUT}/{dst}/images", exist_ok=True)
        os.makedirs(f"{OUT}/{dst}/labels", exist_ok=True)
        for img in glob.glob(f"{img_dir}/*"):
            base, ext = os.path.splitext(os.path.basename(img))
            lbl = f"{loc}/{split}/labels/{base}.txt"
            out = []
            if os.path.exists(lbl):
                for line in open(lbl):
                    v = line.split()
                    if len(v) < 5:
                        continue
                    new = MAPS[i][int(v[0])]
                    if new in (None, "UNKNOWN"):
                        continue
                    nums = list(map(float, v[1:]))
                    if len(nums) > 4:
                        xs, ys = nums[0::2], nums[1::2]
                        x1, x2, y1, y2 = min(xs), max(xs), min(ys), max(ys)
                        nums = [(x1 + x2) / 2, (y1 + y2) / 2, x2 - x1, y2 - y1]
                    out.append(f"{FINAL.index(new)} " + " ".join(f"{n:.6f}" for n in nums))
                    counts[new] += 1
            shutil.copy(img, f"{OUT}/{dst}/images/ds{i}_{base}{ext}")
            open(f"{OUT}/{dst}/labels/ds{i}_{base}.txt", "w").write("\n".join(out))

# make sure at least 20% of all photos are in validation
os.makedirs(f"{OUT}/valid/images", exist_ok=True); os.makedirs(f"{OUT}/valid/labels", exist_ok=True)
tr = glob.glob(f"{OUT}/train/images/*"); va = glob.glob(f"{OUT}/valid/images/*")
need = int(0.2 * (len(tr) + len(va))) - len(va)
random.seed(0)
for img in random.sample(tr, max(0, need)):
    b = os.path.splitext(os.path.basename(img))[0]
    shutil.move(img, f"{OUT}/valid/images/")
    shutil.move(f"{OUT}/train/labels/{b}.txt", f"{OUT}/valid/labels/")

yaml.safe_dump({"path": OUT, "train": "train/images", "val": "valid/images", "names": FINAL},
               open(f"{OUT}/data.yaml", "w"))
print("train:", len(glob.glob(f"{OUT}/train/images/*")), " valid:", len(glob.glob(f"{OUT}/valid/images/*")))
print("boxes per class:", dict(counts))
print("CELL 4 DONE")
